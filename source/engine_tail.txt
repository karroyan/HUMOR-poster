# One scene feeds SVG, print PDF and the native Figma importer.
scene={'width':W,'height':H,'nodes':nodes}
(ROOT/'HUMOR-scene.json').write_text(json.dumps(scene,ensure_ascii=False),encoding='utf8')
fonts=Path('/System/Library/Fonts/Supplemental')
pdfmetrics.registerFont(TTFont('Arial',str(fonts/'Arial.ttf')))
pdfmetrics.registerFont(TTFont('Arial-Bold',str(fonts/'Arial Bold.ttf')))
scale=72/25.4
c=canvas.Canvas(str(ROOT/'HUMOR-poster-print.pdf'),pagesize=(W*scale,H*scale))
c.setTitle('HUMOR | From Perception to Punchline | NeurIPS 2026')
c.setAuthor('Xueyan Li, Yingyi Xue, Mengjie Jiang, Qingzi Zhu, Yazhe Niu')
c.scale(scale,scale)
def color(s):
    return tuple(int(s[i:i+2],16)/255 for i in [1,3,5])
svg=[f'<svg xmlns="http://www.w3.org/2000/svg" xmlns:xlink="http://www.w3.org/1999/xlink" width="{W*96/25.4:.4f}" height="{H*96/25.4:.4f}" viewBox="0 0 {W} {H}">', '<title>HUMOR — From Perception to Punchline</title>']
last=None
for ni,nd in enumerate(nodes):
    if nd['section']!=last:
        if last is not None: svg.append('</g>')
        last=nd['section']; svg.append(f'<g id="{html.escape(last,quote=True)}">')
    t=nd['type']; name='layer-'+str(ni)
    if t=='rect':
        x,y,w,h= [nd[k] for k in ['x','y','w','h']]
        c.setFillColorRGB(*color(nd['fill'])); stroke=nd['stroke']
        if stroke: c.setStrokeColorRGB(*color(stroke)); c.setLineWidth(nd['strokeWidth'])
        c.roundRect(x,H-y-h,w,h,nd['radius'],fill=1,stroke=int(bool(stroke)))
        svg.append(f'<rect id="{name}" x="{x}" y="{y}" width="{w}" height="{h}" rx="{nd["radius"]}" fill="{nd["fill"]}" stroke="{stroke or "none"}" stroke-width="{nd["strokeWidth"]}"/>')
    elif t=='line':
        c.setStrokeColorRGB(*color(nd['stroke'])); c.setLineWidth(nd['strokeWidth'])
        c.line(nd['x1'],H-nd['y1'],nd['x2'],H-nd['y2'])
        svg.append(f'<line x1="{nd["x1"]}" y1="{nd["y1"]}" x2="{nd["x2"]}" y2="{nd["y2"]}" stroke="{nd["stroke"]}" stroke-width="{nd["strokeWidth"]}"/>')
    elif t=='text':
        f='Arial-Bold' if nd['weight']=='bold' else 'Arial'
        c.setFont(f,nd['size']); c.setFillColorRGB(*color(nd['fill']))
        draw={'left':c.drawString,'center':c.drawCentredString,'right':c.drawRightString}[nd['align']]
        draw(nd['x'],H-nd['y'],nd['text'])
        anchor={'left':'start','center':'middle','right':'end'}[nd['align']]
        svg.append(f'<text id="{name}" x="{nd["x"]}" y="{nd["y"]}" font-family="Arial" font-size="{nd["size"]}" font-weight="{700 if nd["weight"]=="bold" else 400}" text-anchor="{anchor}" fill="{nd["fill"]}">{html.escape(nd["text"])}</text>')
    elif t=='image':
        import io
        data=base64.b64decode(nd['data']); c.drawImage(ImageReader(io.BytesIO(data)),nd['x'],H-nd['y']-nd['h'],nd['w'],nd['h'],mask='auto')
        svg.append(f'<image id="{name}" x="{nd["x"]}" y="{nd["y"]}" width="{nd["w"]}" height="{nd["h"]}" xlink:href="data:image/png;base64,{nd["data"]}"/>')
svg.extend(['</g>','</svg>']); (ROOT/'HUMOR-poster-figma.svg').write_text('\n'.join(svg),encoding='utf8')
c.save()
(ROOT/'source'/'poster-copy.txt').write_text('\n'.join(n['text'] for n in nodes if n['type']=='text'),encoding='utf8')
